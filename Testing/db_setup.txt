CREATE TABLE users (
  uid int(11) NOT NULL AUTO_INCREMENT,
  fbid varchar(255) DEFAULT NULL,
  verified tinyint(1) DEFAULT NULL,
  firstName varchar(255) DEFAULT NULL,
  lastName varchar(255) DEFAULT NULL,
  email varchar(255) DEFAULT NULL,
  password varchar(255) DEFAULT NULL,
  account int(11) DEFAULT NULL,
  services varchar(255) DEFAULT NULL,
  description varchar(255) DEFAULT NULL,
  enabled tinyint(1) DEFAULT NULL,
  region varchar(255) DEFAULT NULL,
  city varchar(255) DEFAULT NULL,
  format int(11) DEFAULT NULL,
  createdAt datetime NOT NULL,
  updatedAt datetime NOT NULL,
  PRIMARY KEY (uid)
) ENGINE=InnoDB AUTO_INCREMENT=1 DEFAULT CHARSET=utf8mb4;

CREATE TABLE appointments (
  aid int(11) NOT NULL AUTO_INCREMENT,
  healer varchar(255) DEFAULT NULL,
  client varchar(255) DEFAULT NULL,
  timezone varchar(255) DEFAULT NULL,
  date varchar(255) DEFAULT NULL,
  time varchar(255) DEFAULT NULL,
  healerAccepted tinyint(1) DEFAULT NULL,
  createdAt datetime NOT NULL,
  updatedAt datetime NOT NULL,
  PRIMARY KEY (aid)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE availability (
  id int(11) NOT NULL AUTO_INCREMENT,
  healer varchar(255) DEFAULT NULL,
  timeslots varchar(255) DEFAULT NULL,
  timezone varchar(255) DEFAULT NULL,
  duration varchar(255) DEFAULT NULL,
  createdAt datetime NOT NULL,
  updatedAt datetime NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE images (
  uid int(11) NOT NULL,
  type varchar(255) DEFAULT NULL,
  name varchar(255) NOT NULL,
  data longblob DEFAULT NULL,
  createdAt datetime NOT NULL,
  updatedAt datetime NOT NULL,
  PRIMARY KEY (uid,name),
  CONSTRAINT images_ibfk_1 FOREIGN KEY (uid) REFERENCES users (uid)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE locations (
  lid int(11) NOT NULL AUTO_INCREMENT,
  uid int(11) NOT NULL,
  fbid varchar(255) DEFAULT NULL,
  address varchar(255) DEFAULT NULL,
  lat float(13,10) DEFAULT NULL,
  lng float(13,10) DEFAULT NULL,
  createdAt datetime NOT NULL,
  updatedAt datetime NOT NULL,
  PRIMARY KEY (lid,uid),
  KEY uid (uid),
  CONSTRAINT locations_ibfk_1 FOREIGN KEY (uid) REFERENCES users (uid)
) ENGINE=InnoDB AUTO_INCREMENT=4 DEFAULT CHARSET=utf8mb4;

CREATE TABLE reviews (
  rid int(11) NOT NULL AUTO_INCREMENT,
  reviewer varchar(255) DEFAULT NULL,
  reviewee int(11) DEFAULT NULL,
  rating int(11) DEFAULT NULL,
  comment varchar(255) DEFAULT NULL,
  createdAt datetime NOT NULL,
  updatedAt datetime NOT NULL,
  PRIMARY KEY (rid),
  KEY reviewee (reviewee),
  CONSTRAINT reviews_ibfk_1 FOREIGN KEY (reviewee) REFERENCES users (uid)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE services (
  sid int(11) NOT NULL AUTO_INCREMENT,
  uid int(11) DEFAULT NULL,
  service varchar(255) DEFAULT NULL,
  description varchar(255) DEFAULT NULL,
  createdAt datetime NOT NULL,
  updatedAt datetime NOT NULL,
  PRIMARY KEY (sid)
) ENGINE=InnoDB AUTO_INCREMENT=5 DEFAULT CHARSET=utf8mb4;

CREATE TABLE socials (
  uid int(11) NOT NULL,
  type varchar(255) DEFAULT NULL,
  tag varchar(255) NOT NULL,
  createdAt datetime NOT NULL,
  updatedAt datetime NOT NULL,
  PRIMARY KEY (uid,tag),
  CONSTRAINT socials_ibfk_1 FOREIGN KEY (uid) REFERENCES users (uid)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
